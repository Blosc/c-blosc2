cmake_minimum_required(VERSION 3.16)
project(exec_plan_contract LANGUAGES C)

# Small artifact-off fixture for iteration without building storage codecs.
# The installed-miniexpr fixture separately checks integration/static linking.
find_package(miniexpr CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(exec_plan ../installed-miniexpr/exec-plan.c ../../blosc/b2nd-exec.c)
target_include_directories(exec_plan PRIVATE ../../blosc)
target_link_libraries(exec_plan PRIVATE miniexpr::core ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(exec_plan PRIVATE /W4 /WX)
else()
    target_compile_options(exec_plan PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME exec_plan COMMAND exec_plan)
option(EXEC_PLAN_TEST_JIT "Require a working native JIT compiler in the fixture" OFF)
if(EXEC_PLAN_TEST_JIT)
    add_test(NAME exec_plan_require_jit COMMAND exec_plan --require-jit)
endif()
add_executable(exec_plan_faults faults.c)
target_include_directories(exec_plan_faults PRIVATE ../../blosc)
target_link_libraries(exec_plan_faults PRIVATE miniexpr::core ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(exec_plan_faults PRIVATE /W4 /WX)
else()
    target_compile_options(exec_plan_faults PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME exec_plan_faults COMMAND exec_plan_faults)
add_executable(exec_plan_launcher launch.c)
target_include_directories(exec_plan_launcher PRIVATE ../../blosc)
target_link_libraries(exec_plan_launcher PRIVATE miniexpr::core ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(exec_plan_launcher PRIVATE /W4 /WX)
else()
    target_compile_options(exec_plan_launcher PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME exec_plan_launcher COMMAND exec_plan_launcher)
set_tests_properties(exec_plan_launcher PROPERTIES SKIP_RETURN_CODE 77)
add_executable(exec_host_launcher host-launch.c)
target_include_directories(exec_host_launcher PRIVATE ../../blosc ../../include)
target_link_libraries(exec_host_launcher PRIVATE miniexpr::core ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(exec_host_launcher PRIVATE /W4 /WX)
else()
    target_compile_options(exec_host_launcher PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME exec_host_launcher COMMAND exec_host_launcher)
foreach(target exec_plan exec_plan_faults exec_plan_launcher exec_host_launcher)
    target_link_libraries(${target} PRIVATE Threads::Threads)
endforeach()
