cmake_minimum_required(VERSION 3.16)
project(installed_miniexpr_consumer LANGUAGES C)

find_package(Blosc2 CONFIG REQUIRED)
if(NOT TARGET miniexpr::core)
    message(FATAL_ERROR "Blosc2 did not resolve its static miniexpr dependency")
endif()
add_executable(installed_miniexpr_consumer main.c)
# Prefer the candidate's headers even if a system Blosc2 is already installed.
set_property(TARGET installed_miniexpr_consumer PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_link_libraries(installed_miniexpr_consumer PRIVATE Blosc2::blosc2_static)
enable_testing()
add_test(NAME installed_miniexpr_consumer COMMAND installed_miniexpr_consumer)

add_executable(installed_exec_plan exec-plan.c)
set_property(TARGET installed_exec_plan PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_include_directories(installed_exec_plan PRIVATE ../../blosc)
target_link_libraries(installed_exec_plan PRIVATE Blosc2::blosc2_static miniexpr::core)
add_test(NAME installed_exec_plan COMMAND installed_exec_plan)

add_executable(installed_exec_storage exec-storage.c)
set_property(TARGET installed_exec_storage PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_include_directories(installed_exec_storage PRIVATE ../../blosc)
target_link_libraries(installed_exec_storage PRIVATE Blosc2::blosc2_static miniexpr::core)
add_test(NAME installed_exec_storage COMMAND installed_exec_storage)

add_executable(installed_exec_launcher exec-launch.c)
set_property(TARGET installed_exec_launcher PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_include_directories(installed_exec_launcher PRIVATE ../../blosc)
target_link_libraries(installed_exec_launcher PRIVATE Blosc2::blosc2_static miniexpr::core)
add_test(NAME installed_exec_launcher COMMAND installed_exec_launcher)
set_tests_properties(installed_exec_launcher PROPERTIES SKIP_RETURN_CODE 77)

add_executable(installed_exec_stages exec-stages.c)
set_property(TARGET installed_exec_stages PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_include_directories(installed_exec_stages PRIVATE ../../blosc)
target_link_libraries(installed_exec_stages PRIVATE Blosc2::blosc2_static miniexpr::core)
add_test(NAME installed_exec_stages COMMAND installed_exec_stages)

add_executable(installed_public_exec public-exec.c)
set_property(TARGET installed_public_exec PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_link_libraries(installed_public_exec PRIVATE Blosc2::blosc2_static)
add_test(NAME installed_public_exec COMMAND installed_public_exec)

add_executable(installed_public_exec_kernel public-exec-kernel.c)
set_property(TARGET installed_public_exec_kernel PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_link_libraries(installed_public_exec_kernel PRIVATE Blosc2::blosc2_static miniexpr::core)
add_test(NAME installed_public_exec_kernel COMMAND installed_public_exec_kernel)
