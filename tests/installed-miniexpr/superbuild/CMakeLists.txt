cmake_minimum_required(VERSION 3.16)
project(parent_miniexpr_consumer LANGUAGES C)

set(MINIEXPR_BUILD_SHARED OFF CACHE BOOL "")
set(MINIEXPR_BUILD_STATIC ON CACHE BOOL "")
set(MINIEXPR_BUILD_ARTIFACT OFF CACHE BOOL "")
set(MINIEXPR_INSTALL_CORE ON CACHE BOOL "")
set(MINIEXPR_BUILD_TESTS OFF CACHE BOOL "")
set(MINIEXPR_BUILD_EXAMPLES OFF CACHE BOOL "")
set(MINIEXPR_BUILD_BENCH OFF CACHE BOOL "")
set(MINIEXPR_ENABLE_TCC_JIT OFF CACHE BOOL "")
set(MINIEXPR_USE_SLEEF OFF CACHE BOOL "")
add_subdirectory("${MINIEXPR_SOURCE_DIR}" miniexpr)

set(BLOSC2_USE_PARENT_MINIEXPR ON CACHE BOOL "")
set(BUILD_SHARED OFF CACHE BOOL "")
set(BUILD_STATIC ON CACHE BOOL "")
set(BUILD_TESTS OFF CACHE BOOL "")
set(BUILD_FUZZERS OFF CACHE BOOL "")
set(BUILD_BENCHMARKS OFF CACHE BOOL "")
set(BUILD_EXAMPLES OFF CACHE BOOL "")
get_filename_component(blosc_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${blosc_source}" blosc2)

add_executable(parent_exec_plan ../exec-plan.c)
target_include_directories(parent_exec_plan PRIVATE "${blosc_source}/include" "${blosc_source}/blosc")
target_link_libraries(parent_exec_plan PRIVATE blosc2_static miniexpr::core)
enable_testing()
add_test(NAME parent_exec_plan COMMAND parent_exec_plan)
