cmake_minimum_required(VERSION 3.16)
project(exec_stages_contract LANGUAGES C)
find_package(Blosc2 CONFIG REQUIRED)
add_executable(exec_stages ../installed-miniexpr/exec-stages.c
    ../../blosc/b2nd-exec-stages.c ../../blosc/b2nd-exec.c)
add_executable(exec_stages_faults faults.c ../../blosc/b2nd-exec.c)
foreach(target exec_stages exec_stages_faults)
    set_property(TARGET ${target} PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
    target_include_directories(${target} PRIVATE ../../blosc)
    target_link_libraries(${target} PRIVATE Blosc2::blosc2_static miniexpr::core)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME exec_stages COMMAND exec_stages)
add_test(NAME exec_stages_faults COMMAND exec_stages_faults)

# Optional comparison only. The ordinary native fixture remains artifact-off.
set(EXEC_STAGES_ADAPTER_LIBRARY "" CACHE FILEPATH "Qualification-only miniexpr artifact archive")
set(EXEC_STAGES_ADAPTER_INCLUDE_DIR "" CACHE PATH "Qualification-only miniexpr graph headers")
if(EXEC_STAGES_ADAPTER_LIBRARY AND EXEC_STAGES_ADAPTER_INCLUDE_DIR)
    add_executable(exec_stages_adapter ../installed-miniexpr/exec-stages.c
        ../../blosc/b2nd-exec-stages.c ../../blosc/b2nd-exec.c)
    set_property(TARGET exec_stages_adapter PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
    target_compile_definitions(exec_stages_adapter PRIVATE STAGE_ADAPTER_TEST=1)
    if(MSVC)
        target_compile_options(exec_stages_adapter PRIVATE /W4 /WX)
    else()
        target_compile_options(exec_stages_adapter PRIVATE -Wall -Wextra -Werror)
    endif()
    target_include_directories(exec_stages_adapter PRIVATE ../../blosc ${EXEC_STAGES_ADAPTER_INCLUDE_DIR})
    target_link_libraries(exec_stages_adapter PRIVATE ${EXEC_STAGES_ADAPTER_LIBRARY}
        Blosc2::blosc2_static miniexpr::core)
    add_test(NAME exec_stages_adapter COMMAND exec_stages_adapter)
endif()
