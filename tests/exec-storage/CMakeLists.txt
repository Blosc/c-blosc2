cmake_minimum_required(VERSION 3.16)
project(exec_storage_contract LANGUAGES C)
find_package(Blosc2 CONFIG REQUIRED)
add_executable(exec_storage ../installed-miniexpr/exec-storage.c
    ../../blosc/b2nd-exec.c ../../blosc/b2nd-exec-storage.c)
add_executable(exec_storage_faults faults.c ../../blosc/b2nd-exec.c)
foreach(target exec_storage exec_storage_faults)
    set_property(TARGET ${target} PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
    target_include_directories(${target} PRIVATE ../../blosc)
    target_link_libraries(${target} PRIVATE Blosc2::blosc2_static miniexpr::core)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME exec_storage COMMAND exec_storage)
add_test(NAME exec_storage_faults COMMAND exec_storage_faults)
